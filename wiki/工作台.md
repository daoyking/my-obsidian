# Knowledge HUD // my-obsidian

> 自成长知识库 · 卡帕西 LLM Wiki 方法 · 实时状态面板

```dataviewjs
const wp = dv.pages('"wiki"').where(p => p.type != null);
const pp = dv.pages('"projects"');
const raw = dv.pages('"raw"');
const total = wp.length + pp.length;
const active = wp.where(p => p.status == "active").length + pp.where(p => p.status == "active").length;
const seed = wp.where(p => p.status == "seed").length + pp.where(p => p.status == "seed").length;
const concepts = wp.where(p => p.type == "concept").length;
const entities = wp.where(p => p.type == "entity").length;
const topics = wp.where(p => p.type == "topic").length;

dv.table(
  ["指标", "概念", "实体", "主题", "总页", "ACTIVE", "SEED", "RAW源"],
  [["**数量**", `**${concepts}**`, `**${entities}**`, `**${topics}**`, `**${total}**`, `**${active}**`, `**${seed}**`, `**${raw.length}**`]]
);
```

## [ 最近更新 // TOP 12 ]

```dataview
TABLE WITHOUT ID
  file.link AS "页面", type AS "类型", status AS "状态", dateformat(file.mtime, "MM-dd HH:mm") AS "更新"
FROM "wiki" AND -"wiki/INDEX" AND -"wiki/LOG"
WHERE type != null
SORT file.mtime DESC
LIMIT 12
```

---

## [ 待填充 SEED // 缺口 ]

```dataview
TABLE WITHOUT ID file.link AS "页面", type AS "类型"
FROM "wiki"
WHERE status = "seed"
SORT type, file.name
```

---

## [ RAW 原始源 ]

```dataview
TABLE WITHOUT ID file.link AS "源文件", ingested AS "导入日"
FROM "raw"
WHERE type = "source"
SORT ingested DESC
```

---

## [ 概念分布 ]

```dataview
TABLE WITHOUT ID file.link AS "概念", status AS "状态", updated AS "更新"
FROM "wiki/concepts"
SORT status, updated DESC
```
